📖 ESP32 DevKit Wiring & Firmware (with mDNS & Auto-Reboot)

ESP32 Hardware Wiring

The built-in LED on ESP32 DevKit is connected to GPIO 2. For external LED control, connect an LED with a 220Ω resistor between GPIO 2 and GND.

2
Flash Code via Arduino IDE

In Arduino IDE, select ESP32 Dev Module, set Baud to 115200, copy the code below, and click Upload. Open Serial Monitor (115200) to see connection logs.

📄 esp32_voice_assistant.ino 
/*
  ========================================================================
  ESP32 AI Voice Assistant Firmware
  Features:
    - Non-Volatile Preferences storage for Wi-Fi SSID & Password
    - Automatic ESP.restart() after credential updates
    - mDNS support: Access via http://sara.local
    - Dual SoftAP + Station WiFi Mode with auto-reconnect fallback
    - Web Server REST API (/status, /led/on, /led/off, /led/toggle, /wifi/save, /reboot)
    - USB Web Serial command processor (115200 Baud)
  ========================================================================
*/
#include <WiFi.h>
#include <WebServer.h>
#include <Preferences.h>
#include <ESPmDNS.h>

#define LED_PIN 2

WebServer server(80);
Preferences preferences;

bool ledState = false;
String wifiSSID = "";
String wifiPASS = "";
unsigned long bootTime = 0;

void setLed(bool state) {
  ledState = state;
  digitalWrite(LED_PIN, ledState ? HIGH : LOW);
}

void sendCORSHeader() {
  server.sendHeader("Access-Control-Allow-Origin", "*");
  server.sendHeader("Access-Control-Allow-Methods", "GET, POST, OPTIONS");
  server.sendHeader("Access-Control-Allow-Headers", "*");
}

void handleStatus() {
  sendCORSHeader();
  unsigned long uptimeSec = (millis() - bootTime) / 1000;
  int rssi = WiFi.status() == WL_CONNECTED ? WiFi.RSSI() : 0;
  String ipStr = WiFi.status() == WL_CONNECTED ? WiFi.localIP().toString() : WiFi.softAPIP().toString();

  String json = "{";
  json += "\"status\":\"online\",";
  json += "\"led\":" + String(ledState ? 1 : 0) + ",";
  json += "\"wifi_connected\":" + String(WiFi.status() == WL_CONNECTED ? "true" : "false") + ",";
  json += "\"ip\":\"" + ipStr + "\",";
  json += "\"rssi\":" + String(rssi) + ",";
  json += "\"uptime\":" + String(uptimeSec);
  json += "}";

  server.send(200, "application/json", json);
}

void handleLedOn() {
  setLed(true);
  handleStatus();
}

void handleLedOff() {
  setLed(false);
  handleStatus();
}

void handleLedToggle() {
  setLed(!ledState);
  handleStatus();
}

void handleReboot() {
  sendCORSHeader();
  server.send(200, "application/json", "{\"status\":\"rebooting\",\"message\":\"ESP32 is restarting...\"}");
  delay(1000);
  ESP.restart();
}

void handleWifiSave() {
  sendCORSHeader();
  if (server.hasArg("ssid")) {
    wifiSSID = server.arg("ssid");
    wifiPASS = server.hasArg("pass") ? server.arg("pass") : "";

    preferences.begin("wifi_config", false);
    preferences.putString("ssid", wifiSSID);
    preferences.putString("pass", wifiPASS);
    preferences.end();

    server.send(200, "application/json", "{\"status\":\"saved\",\"message\":\"Credentials saved! ESP32 is restarting now...\"}");
    
    Serial.println("Wi-Fi Credentials Saved. Restarting ESP32...");
    delay(1500);
    ESP.restart();
  } else {
    server.send(400, "application/json", "{\"error\":\"Missing ssid parameter\"}");
  }
}

void setup() {
  Serial.begin(115200);
  pinMode(LED_PIN, OUTPUT);
  digitalWrite(LED_PIN, LOW);
  bootTime = millis();

  // Load saved Wi-Fi credentials from Flash Preferences
  preferences.begin("wifi_config", true);
  wifiSSID = preferences.getString("ssid", "");
  wifiPASS = preferences.getString("pass", "");
  preferences.end();

  // Enable AP Mode for initial setup
  WiFi.mode(WIFI_AP_STA);
  WiFi.softAP("ESP32_Sara_Setup", "12345678");
  Serial.println("------------------------------------------");
  Serial.println("SoftAP Started: ESP32_Sara_Setup (Pass: 12345678)");
  Serial.println("AP IP address: " + WiFi.softAPIP().toString());

  // If Wi-Fi credentials saved, attempt to connect to router
  if (wifiSSID.length() > 0) {
    Serial.println("Connecting to Wi-Fi SSID: " + wifiSSID);
    WiFi.begin(wifiSSID.c_str(), wifiPASS.c_str());

    int attempts = 0;
    while (WiFi.status() != WL_CONNECTED && attempts < 20) {
      delay(500);
      Serial.print(".");
      attempts++;
    }

    if (WiFi.status() == WL_CONNECTED) {
      Serial.println("\n✅ Wi-Fi Connected!");
      Serial.println("Station IP: " + WiFi.localIP().toString());
    } else {
      Serial.println("\n⚠️ Wi-Fi connection timed out. SoftAP remains active.");
    }
  }

  // Start mDNS responder (allows http://sara.local access)
  if (MDNS.begin("sara")) {
    Serial.println("✨ mDNS responder started: http://sara.local");
  }

  // Define HTTP REST endpoints
  server.on("/status", HTTP_GET, handleStatus);
  server.on("/led/on", HTTP_GET, handleLedOn);
  server.on("/led/off", HTTP_GET, handleLedOff);
  server.on("/led/toggle", HTTP_GET, handleLedToggle);
  server.on("/wifi/save", HTTP_GET, handleWifiSave);
  server.on("/reboot", HTTP_GET, handleReboot);

  // Enable CORS preflight OPTIONS request handling
  server.onNotFound([]() {
    if (server.method() == HTTP_OPTIONS) {
      sendCORSHeader();
      server.send(204);
    } else {
      server.send(404, "text/plain", "Not found");
    }
  });

  server.begin();
  Serial.println("🚀 HTTP Web Server Started.");
  Serial.println("------------------------------------------");
}

void loop() {
  server.handleClient();

  // Process incoming USB Serial commands
  if (Serial.available() > 0) {
    String cmd = Serial.readStringUntil('\n');
    cmd.trim();
    cmd.toUpperCase();

    if (cmd == "LED_ON" || cmd == "ON" || cmd == "1") {
      setLed(true);
      Serial.println("STATUS:ON");
    } else if (cmd == "LED_OFF" || cmd == "OFF" || cmd == "0") {
      setLed(false);
      Serial.println("STATUS:OFF");
    } else if (cmd == "LED_TOGGLE" || cmd == "TOGGLE") {
      setLed(!ledState);
      Serial.println(ledState ? "STATUS:ON" : "STATUS:OFF");
    } else if (cmd == "STATUS") {
      Serial.println(ledState ? "STATUS:ON" : "STATUS:OFF");
    } else if (cmd == "REBOOT") {
      Serial.println("Rebooting ESP32...");
      delay(500);
      ESP.restart();
    } else if (cmd.startsWith("WIFI:")) {
      int commaPos = cmd.indexOf(',');
      if (commaPos > 5) {
        String ssid = cmd.substring(5, commaPos);
        String pass = cmd.substring(commaPos + 1);
        preferences.begin("wifi_config", false);
        preferences.putString("ssid", ssid);
        preferences.putString("pass", pass);
        preferences.end();
        Serial.println("WIFI_SAVED:" + ssid + ". Restarting...");
        delay(1000);
        ESP.restart();
      }
    }
  }
}

🔌 Arduino Uno / Nano USB Serial Firmware

If using an Arduino Uno or Nano (ATmega328P without Wi-Fi), connect via USB cable, set baud rate to 115200 or 9600, and upload the sketch below.

📄 arduino_led_control.ino 
/*
  Arduino Uno / Nano AI Voice Assistant - Web Serial Sketch
  Baud rate: 115200
*/
const int LED_PIN = LED_BUILTIN; // Pin 13 on Arduino Uno
bool ledState = false;

void setup() {
  Serial.begin(115200);
  pinMode(LED_PIN, OUTPUT);
  digitalWrite(LED_PIN, LOW);
  Serial.println("ARDUINO UNO VOICE CONTROL READY");
  Serial.println("STATUS:OFF");
}

void loop() {
  if (Serial.available() > 0) {
    String input = Serial.readStringUntil('\n');
    input.trim();
    input.toUpperCase();

    if (input == "1" || input == "ON" || input == "LED_ON") {
      setLedState(true);
    } else if (input == "0" || input == "OFF" || input == "LED_OFF") {
      setLedState(false);
    } else if (input == "TOGGLE" || input == "T" || input == "LED_TOGGLE") {
      setLedState(!ledState);
    } else if (input == "STATUS" || input == "S") {
      reportStatus();
    }
  }
}

void setLedState(bool state) {
  ledState = state;
  digitalWrite(LED_PIN, ledState ? HIGH : LOW);
  reportStatus();
}

void reportStatus() {
  if (ledState) {
    Serial.println("STATUS:ON");
  } else {
    Serial.println("STATUS:OFF");
  }
}
