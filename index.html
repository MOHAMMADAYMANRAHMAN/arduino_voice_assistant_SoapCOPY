Send your Home Wi-Fi or Mobile Hotspot credentials directly to your ESP32 board over Wi-Fi or USB Serial. The board will save credentials to non-volatile flash storage and reboot cleanly.

Step-by-Step Wireless Setup & Auto-Reboot:
1
Power on ESP32

Flash the updated ESP32 firmware from the Firmware Guide tab and power on your board.

2
Connect to Access Point

Connect your computer or phone to Wi-Fi: ESP32_Sara_Setup (Password: 12345678).

3
Send Credentials & Auto Reboot

Set IP address to 192.168.4.1, click Test Connection, enter your SSID/Password above, and submit. The board will reboot and register as http://sara.local on your Wi-Fi!

📖 ESP32 DevKit Firmware (with Item Dispatch & Serial Monitor Printing)

1
ESP32 Hardware Wiring

Built-in LED is on GPIO 2. Connect an external LED with a 220Ω resistor between GPIO 2 and GND.

2
Flash Code via Arduino IDE

In Arduino IDE, select ESP32 Dev Module, set Baud to 115200, copy the code below, and click Upload. Open Serial Monitor (115200) to see connection logs and item dispatches.

📄 esp32_voice_assistant.ino 
/*
  ========================================================================
  ESP32 AI Voice Assistant & Smart Inventory Firmware
  Features:
    - Serial Monitor Item Printer (Category, Item Name, Quantity)
    - /item REST Endpoint & ITEM: Category,Name,Qty Serial Command
    - Non-Volatile Preferences storage for Wi-Fi SSID & Password
    - Automatic ESP.restart() after credential updates
    - mDNS support: Access via http://sara.local
    - Dual SoftAP + Station WiFi Mode
  ========================================================================
*/
#include <WiFi.h>
#include <WebServer.h>
#include <Preferences.h>
#include <ESPmDNS.h>

#define LED_PIN 2

WebServer server(80);
Preferences preferences;

bool ledState = false;
String wifiSSID = "";
String wifiPASS = "";
unsigned long bootTime = 0;

void setLed(bool state) {
  ledState = state;
  digitalWrite(LED_PIN, ledState ? HIGH : LOW);
}

void sendCORSHeader() {
  server.sendHeader("Access-Control-Allow-Origin", "*");
  server.sendHeader("Access-Control-Allow-Methods", "GET, POST, OPTIONS");
  server.sendHeader("Access-Control-Allow-Headers", "*");
}

void printItemToSerial(String category, String name, String qty) {
  Serial.println("\n==================================================");
  Serial.println("📦 ITEM DISPATCHED TO SERIAL MONITOR");
  Serial.println("   Category : " + category);
  Serial.println("   Item Name: " + name);
  Serial.println("   Quantity : " + qty);
  Serial.println("==================================================\n");
}

void handleStatus() {
  sendCORSHeader();
  unsigned long uptimeSec = (millis() - bootTime) / 1000;
  int rssi = WiFi.status() == WL_CONNECTED ? WiFi.RSSI() : 0;
  String ipStr = WiFi.status() == WL_CONNECTED ? WiFi.localIP().toString() : WiFi.softAPIP().toString();

  String json = "{";
  json += "\"status\":\"online\",";
  json += "\"led\":" + String(ledState ? 1 : 0) + ",";
  json += "\"wifi_connected\":" + String(WiFi.status() == WL_CONNECTED ? "true" : "false") + ",";
  json += "\"ip\":\"" + ipStr + "\",";
  json += "\"rssi\":" + String(rssi) + ",";
  json += "\"uptime\":" + String(uptimeSec);
  json += "}";

  server.send(200, "application/json", json);
}

void handleLedOn() {
  setLed(true);
  handleStatus();
}

void handleLedOff() {
  setLed(false);
  handleStatus();
}

void handleLedToggle() {
  setLed(!ledState);
  handleStatus();
}

void handleItem() {
  sendCORSHeader();
  String category = server.hasArg("cat") ? server.arg("cat") : "General";
  String name = server.hasArg("name") ? server.arg("name") : "Unknown Item";
  String qty = server.hasArg("qty") ? server.arg("qty") : "1";

  printItemToSerial(category, name, qty);

  String json = "{\"status\":\"printed\",\"category\":\"" + category + "\",\"name\":\"" + name + "\",\"qty\":\"" + qty + "\"}";
  server.send(200, "application/json", json);
}

void handleReboot() {
  sendCORSHeader();
  server.send(200, "application/json", "{\"status\":\"rebooting\",\"message\":\"ESP32 is restarting...\"}");
  delay(1000);
  ESP.restart();
}

void handleWifiSave() {
  sendCORSHeader();
  if (server.hasArg("ssid")) {
    wifiSSID = server.arg("ssid");
    wifiPASS = server.hasArg("pass") ? server.arg("pass") : "";

    preferences.begin("wifi_config", false);
    preferences.putString("ssid", wifiSSID);
    preferences.putString("pass", wifiPASS);
    preferences.end();

    server.send(200, "application/json", "{\"status\":\"saved\",\"message\":\"Credentials saved! ESP32 is restarting...\"}");
    
    Serial.println("Wi-Fi Credentials Saved. Restarting ESP32...");
    delay(1500);
    ESP.restart();
  } else {
    server.send(400, "application/json", "{\"error\":\"Missing ssid parameter\"}");
  }
}

void setup() {
  Serial.begin(115200);
  pinMode(LED_PIN, OUTPUT);
  digitalWrite(LED_PIN, LOW);
  bootTime = millis();

  preferences.begin("wifi_config", true);
  wifiSSID = preferences.getString("ssid", "");
  wifiPASS = preferences.getString("pass", "");
  preferences.end();

  WiFi.mode(WIFI_AP_STA);
  WiFi.softAP("ESP32_Sara_Setup", "12345678");
  Serial.println("------------------------------------------");
  Serial.println("SoftAP Started: ESP32_Sara_Setup (Pass: 12345678)");
  Serial.println("AP IP address: " + WiFi.softAPIP().toString());

  if (wifiSSID.length() > 0) {
    Serial.println("Connecting to Wi-Fi SSID: " + wifiSSID);
    WiFi.begin(wifiSSID.c_str(), wifiPASS.c_str());

    int attempts = 0;
    while (WiFi.status() != WL_CONNECTED && attempts < 20) {
      delay(500);
      Serial.print(".");
      attempts++;
    }

    if (WiFi.status() == WL_CONNECTED) {
      Serial.println("\n✅ Wi-Fi Connected!");
      Serial.println("Station IP: " + WiFi.localIP().toString());
    } else {
      Serial.println("\n⚠️ Wi-Fi connection timed out. SoftAP remains active.");
    }
  }

  if (MDNS.begin("sara")) {
    Serial.println("✨ mDNS responder started: http://sara.local");
  }

  server.on("/status", HTTP_GET, handleStatus);
  server.on("/led/on", HTTP_GET, handleLedOn);
  server.on("/led/off", HTTP_GET, handleLedOff);
  server.on("/led/toggle", HTTP_GET, handleLedToggle);
  server.on("/item", HTTP_GET, handleItem);
  server.on("/wifi/save", HTTP_GET, handleWifiSave);
  server.on("/reboot", HTTP_GET, handleReboot);

  server.onNotFound([]() {
    if (server.method() == HTTP_OPTIONS) {
      sendCORSHeader();
      server.send(204);
    } else {
      server.send(404, "text/plain", "Not found");
    }
  });

  server.begin();
  Serial.println("🚀 HTTP Web Server Started.");
  Serial.println("------------------------------------------");
}

void loop() {
  server.handleClient();

  if (Serial.available() > 0) {
    String cmd = Serial.readStringUntil('\n');
    cmd.trim();

    if (cmd.equalsIgnoreCase("LED_ON") || cmd.equalsIgnoreCase("ON") || cmd == "1") {
      setLed(true);
      Serial.println("STATUS:ON");
    } else if (cmd.equalsIgnoreCase("LED_OFF") || cmd.equalsIgnoreCase("OFF") || cmd == "0") {
      setLed(false);
      Serial.println("STATUS:OFF");
    } else if (cmd.equalsIgnoreCase("LED_TOGGLE") || cmd.equalsIgnoreCase("TOGGLE")) {
      setLed(!ledState);
      Serial.println(ledState ? "STATUS:ON" : "STATUS:OFF");
    } else if (cmd.startsWith("ITEM:")) {
      // Format: ITEM:Category,Name,Quantity
      String data = cmd.substring(5);
      int c1 = data.indexOf(',');
      int c2 = data.indexOf(',', c1 + 1);
      if (c1 > 0) {
        String cat = data.substring(0, c1);
        String name = (c2 > 0) ? data.substring(c1 + 1, c2) : data.substring(c1 + 1);
        String qty = (c2 > 0) ? data.substring(c2 + 1) : "1";
        printItemToSerial(cat, name, qty);
      }
    } else if (cmd.equalsIgnoreCase("REBOOT")) {
      Serial.println("Rebooting ESP32...");
      delay(500);
      ESP.restart();
    }
  }
}

/*
  Arduino Uno / Nano AI Voice Assistant & Item Dispatch Sketch
*/
const int LED_PIN = LED_BUILTIN;
bool ledState = false;

void printItemToSerial(String category, String name, String qty) {
  Serial.println(F("\n=================================================="));
  Serial.println(F("📦 ITEM DISPATCHED TO SERIAL MONITOR"));
  Serial.print(F("   Category : ")); Serial.println(category);
  Serial.print(F("   Item Name: ")); Serial.println(name);
  Serial.print(F("   Quantity : ")); Serial.println(qty);
  Serial.println(F("==================================================\n"));
}

void setup() {
  Serial.begin(115200);
  pinMode(LED_PIN, OUTPUT);
  digitalWrite(LED_PIN, LOW);
  Serial.println("ARDUINO VOICE & ITEM DISPATCH CONTROL READY");
  Serial.println("STATUS:OFF");
}

void loop() {
  if (Serial.available() > 0) {
    String input = Serial.readStringUntil('\n');
    input.trim();

    if (input == "1" || input.equalsIgnoreCase("ON") || input.equalsIgnoreCase("LED_ON")) {
      setLedState(true);
    } else if (input == "0" || input.equalsIgnoreCase("OFF") || input.equalsIgnoreCase("LED_OFF")) {
      setLedState(false);
    } else if (input.equalsIgnoreCase("TOGGLE") || input.equalsIgnoreCase("LED_TOGGLE")) {
      setLedState(!ledState);
    } else if (input.startsWith("ITEM:")) {
      String data = input.substring(5);
      int c1 = data.indexOf(',');
      int c2 = data.indexOf(',', c1 + 1);
      if (c1 > 0) {
        String cat = data.substring(0, c1);
        String name = (c2 > 0) ? data.substring(c1 + 1, c2) : data.substring(c1 + 1);
        String qty = (c2 > 0) ? data.substring(c2 + 1) : "1";
        printItemToSerial(cat, name, qty);
      }
    }
  }
}

void setLedState(bool state) {
  ledState = state;
  digitalWrite(LED_PIN, ledState ? HIGH : LOW);
  Serial.println(ledState ? "STATUS:ON" : "STATUS:OFF");
}
