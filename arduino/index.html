🚀 Quick Setup Guide

  1. Connect your Arduino Uno to your computer using a USB Type-B cable.
  2. Copy the C++ Sketch shown on the right panel into the Arduino IDE.
  3. In Arduino IDE, select Tools > Board > Arduino Uno and choose your COM port under Tools > Port.
  4. Click Upload (Ctrl+U) to load the code onto your Arduino Uno.
  5. Close the Arduino IDE Serial Monitor (to free the COM port), return to this Web Assistant, and click 🔌 Connect Arduino in the top header bar!
💡 Tip: Browser Web Serial API requires Google Chrome, Microsoft Edge, or Opera. If you are on an unsupported browser, you can use our included serial_bridge.py Python fallback script.

/*
  Arduino Uno AI Voice Assistant - LED Control Sketch
*/
const int LED_PIN = LED_BUILTIN; // Pin 13 on Arduino Uno
bool ledState = false;

void setup() {
  Serial.begin(9600);
  pinMode(LED_PIN, OUTPUT);
  digitalWrite(LED_PIN, LOW);

  // Ready banner
  Serial.println("ARDUINO UNO VOICE CONTROL READY");
  Serial.println("STATUS:OFF");
}

void loop() {
  if (Serial.available() > 0) {
    String input = Serial.readStringUntil('\n');
    input.trim();
    input.toUpperCase();

    if (input == "1" || input == "ON" || input == "LED_ON") {
      setLedState(true);
    } else if (input == "0" || input == "OFF" || input == "LED_OFF") {
      setLedState(false);
    } else if (input == "TOGGLE" || input == "T") {
      setLedState(!ledState);
    } else if (input == "STATUS" || input == "S") {
      reportStatus();
    }
  }
}

void setLedState(bool state) {
  ledState = state;
  digitalWrite(LED_PIN, ledState ? HIGH : LOW);
  reportStatus();
}

void reportStatus() {
  if (ledState) {
    Serial.println("STATUS:ON");
    Serial.println("ACK: LED turned ON");
  } else {
    Serial.println("STATUS:OFF");
    Serial.println("ACK: LED turned OFF");
  }
}
